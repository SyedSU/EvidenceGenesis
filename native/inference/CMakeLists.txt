cmake_minimum_required(VERSION 3.31)
project(genesis_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
foreach(option LLAMA_BUILD_TESTS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_COMMON LLAMA_BUILD_TOOLS LLAMA_BUILD_MTMD LLAMA_OPENSSL GGML_OPENMP GGML_OPENMP_FETCH GGML_LLAMAFILE GGML_NATIVE GGML_CPU_KLEIDIAI GGML_BACKEND_DL GGML_RPC GGML_CUDA GGML_VULKAN GGML_OPENCL GGML_METAL GGML_SYCL GGML_HIP GGML_BLAS)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
add_subdirectory("${GENESIS_LLAMA_SOURCE}" llama)
add_library(genesis_ai SHARED inference_core.cpp inference_extension.c)
target_include_directories(genesis_ai PRIVATE ../durability)
target_link_libraries(genesis_ai PRIVATE llama)
target_compile_options(genesis_ai PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
set_target_properties(genesis_ai PROPERTIES BUILD_RPATH "$ORIGIN/bin" BUILD_RPATH_USE_ORIGIN TRUE INSTALL_RPATH "")
if(ANDROID)
    set_target_properties(genesis_ai PROPERTIES SKIP_BUILD_RPATH TRUE)
    target_link_options(genesis_ai PRIVATE -Wl,--no-undefined -Wl,-z,max-page-size=16384)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(genesis_ai PRIVATE Threads::Threads)
endif()
